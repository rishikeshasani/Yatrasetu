  <script>
    // --- 1. REQUIREMENT 1: GENUINELY DYNAMIC QR GENERATOR ---
    // Encodes arbitrary payload strings using QRCode.js with safe fallback
    function renderDynamicQR(containerId, payload) {
      const container = document.getElementById(containerId);
      if (!container) return;
      container.innerHTML = '';

      if (typeof QRCode !== 'undefined') {
        new QRCode(container, {
          text: payload,
          width: 176,
          height: 176,
          colorDark: "#0f172a",
          colorLight: "#ffffff",
          correctLevel: QRCode.CorrectLevel.M
        });
      } else {
        // Fallback in case of local offline sandbox without cached CDN
        container.innerHTML = `<div class="flex flex-col items-center justify-center p-4 bg-slate-900 text-white rounded-xl border border-slate-700 min-h-[176px] min-w-[176px]">
          <span class="text-3xl mb-1">📱</span>
          <span class="text-xs font-mono font-bold text-amber-400">${payload}</span>
          <span class="text-[10px] text-slate-400 mt-1">Live Encoded Payload</span>
        </div>`;
      }

      document.getElementById('qrBookingPayload').innerText = payload;

    function triggerReroutingSpike() {
      if (state.isRerouteSpikeActive) return;

      state.isRerouteSpikeActive = true;
      state.demandLevel = 'HIGH_SURGE';
      state.suggestedRate = 1300;

      animateCounter('incomingTouristsCount', 120, 180, 600);
      dismissFraudAlert();
      renderDashboard();
    }

    function animateCounter(elementId, start, end, duration) {
      const obj = document.getElementById(elementId);
      let startTimestamp = null;
      const step = (timestamp) => {
        if (!startTimestamp) startTimestamp = timestamp;
        const progress = Math.min((timestamp - startTimestamp) / duration, 1);
        obj.innerHTML = Math.floor(progress * (end - start) + start);
        if (progress < 1) {
          window.requestAnimationFrame(step);
        }
      };
      window.requestAnimationFrame(step);
    }

    function toggleSuggestedRate() {
      state.isSuggestedApplied = !state.isSuggestedApplied;
      state.currentRate = state.isSuggestedApplied ? state.suggestedRate : state.baseRate;
      renderDashboard();
    }

    // --- 6. FLOW 2: GUEST CHECK-IN & UPGRADED COOLDOWN FRAUD (Requirement 2) ---
    function simulateGuestCheckIn() {
      const now = Date.now();
      const lastScanTime = scanCooldownLog[state.bookingRef];

      // Requirement 2: Strict 60-second cooldown rule (blocks duplicate scans even if checked out)
      if (lastScanTime && (now - lastScanTime < COOLDOWN_WINDOW_MS)) {
        triggerFraudAlert("⚠️ Fraud Rule: Duplicate scan detected within cooldown window.");
        return;
      }

      // Record this valid scan timestamp into the log
      scanCooldownLog[state.bookingRef] = now;

      dismissFraudAlert();
      dismissSuccessBanner();

      state.guestStatus = 'CHECKED_IN';
      // Occupancy increases: 32 -> 33 (Available reduces: 18 -> 17)
      state.occupiedRooms = 33;

      renderDashboard();
    }

    // --- 7. FLOW 2B: GUEST CHECK-OUT ---
    function simulateGuestCheckOut() {
      if (state.guestStatus !== 'CHECKED_IN') return;

      state.guestStatus = 'CHECKED_OUT';
      // Available rooms return to 18 (Occupied returns to 32)
      state.occupiedRooms = 32;

      // Award +50 Punya points (1450 -> 1500)
      state.punyaPoints += 50;
      // Hotel Tax Credit increased to ₹750 (725 -> 750)
      state.taxCredit = 750;

      renderDashboard();
      showSuccessBanner();
    }

    // --- 8. FRAUD & SUCCESS BANNER CONTROLLERS ---
    function triggerFraudAlert(msg) {
      const banner = document.getElementById('fraudAlertBanner');
      const text = document.getElementById('fraudMessageText');
      if (msg) text.innerText = msg;
      banner.classList.remove('hidden');
      banner.classList.add('animate-bounce');
      setTimeout(() => banner.classList.remove('animate-bounce'), 800);
    }

    function dismissFraudAlert() {
      document.getElementById('fraudAlertBanner').classList.add('hidden');
    }

    function showSuccessBanner() {
      const banner = document.getElementById('successCheckoutBanner');
      banner.classList.remove('hidden');
    }

    function dismissSuccessBanner() {
      document.getElementById('successCheckoutBanner').classList.add('hidden');
    }
